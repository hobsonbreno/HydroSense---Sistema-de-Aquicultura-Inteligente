# CMakeLists.txt para HydroSense v5 com WiFi
cmake_minimum_required(VERSION 3.13)

set(PICO_BOARD pico_w)

include(pico_sdk_import.cmake)

project(hydrosense_v5 C CXX ASM)
set(CMAKE_C_STANDARD 11)
set(CMAKE_CXX_STANDARD 17)

pico_sdk_init()

add_executable(hydrosense_v5
    src/hydrosense_v5.c
)

target_include_directories(hydrosense_v5 PRIVATE
    ${CMAKE_CURRENT_LIST_DIR}
)

target_link_libraries(hydrosense_v5
    pico_stdlib
    pico_cyw43_arch_lwip_poll
    hardware_i2c
    hardware_gpio
    hardware_pwm
)

pico_enable_stdio_usb(hydrosense_v5 1)
pico_enable_stdio_uart(hydrosense_v5 0)

pico_add_extra_outputs(hydrosense_v5)
